def gmail_access_token(cfg):
    if not cfg['gmail_client_id'] or not cfg['gmail_client_secret'] or not cfg['gmail_refresh_token']:
        return None, 'Gmail not configured yet. Open SETUP and add your OAuth values.'
    resp = requests.post('https://oauth2.googleapis.com/token', data={
        'client_id': cfg['gmail_client_id'],
        'client_secret': cfg['gmail_client_secret'],
        'refresh_token': cfg['gmail_refresh_token'],
        'grant_type': 'refresh_token',
    }, timeout=20)
    if resp.status_code != 200:
        return None, 'Gmail auth failed: ' + resp.text[:150]
    return resp.json().get('access_token'), ''


def extract_body(payload):
    queue = [payload] if payload else []
    while queue:
        part = queue.pop(0)
        if part.get('mimeType') == 'text/plain' and part.get('body', {}).get('data'):
            return base64.urlsafe_b64decode(part['body']['data']).decode('utf-8', 'replace')
        for sub in part.get('parts') or []:
            queue.append(sub)
    return '(no plain-text body)'


def read_emails(cfg, count):
    token, err = gmail_access_token(cfg)
    if not token:
        return err
    headers = {'Authorization': 'Bearer ' + token}
    listing = requests.get(
        'https://gmail.googleapis.com/gmail/v1/users/me/messages',
        params={'maxResults': count}, headers=headers, timeout=20).json()
    items = listing.get('messages') or []
    if not items:
        return 'No messages found.'
    out = []
    for item in items[:count]:
        msg = requests.get(
            'https://gmail.googleapis.com/gmail/v1/users/me/messages/' + item['id'],
            params={'format': 'full'}, headers=headers, timeout=20).json()
        heads = {h['name'].lower(): h['value'] for h in msg.get('payload', {}).get('headers', [])}
        body = extract_body(msg.get('payload', {}))
        out.append('FROM: ' + heads.get('from', '?') +
                   '\nSUBJECT: ' + heads.get('subject', '(none)') +
                   '\n' + body[:300] + '\n' + ('-' * 32))
    audit('read_emails', str(len(out)) + ' messages')
    return '\n'.join(out)


def post_slack(cfg, message, channel):
    if not cfg['slack_token']:
        return 'Slack token not set. Open SETUP first.'
    resp = requests.post('https://slack.com/api/chat.postMessage',
                         headers={'Authorization': 'Bearer ' + cfg['slack_token']},
                         json={'channel': channel, 'text': message}, timeout=20)
    data = resp.json()
    if data.get('ok'):
        audit('post_slack', channel)
        return 'Message posted to ' + channel
    return 'Slack error: ' + data.get('error', 'unknown')


CATEGORIES = {
    'Images': ['.jpg', '.jpeg', '.png', '.gif', '.webp'],
    'Documents': ['.pdf', '.txt', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx'],
    'Audio': ['.mp3', '.wav', '.ogg'],
    'Video': ['.mp4', '.mkv', '.mov'],
    'Archives': ['.zip', '.rar', '.7z'],
}


def organize_files():
    moved = []
    for f in [x for x in WORKSPACE.iterdir() if x.is_file()]:
        cat = 'Other'
        for name, exts in CATEGORIES.items():
            if f.suffix.lower() in exts:
                cat = name
                break
        target = WORKSPACE / cat
        target.mkdir(exist_ok=True)
        dest = target / f.name
        i = 1
        while dest.exists():
            dest = target / (f.stem + '_' + str(i) + f.suffix)
            i += 1
        f.rename(dest)
        moved.append(f.name + '  ->  ' + cat + '/')
    if not moved:
        return 'No loose files in the sandboxed workspace.'
    audit('organize_files', str(len(moved)) + ' files')
    return 'Organized ' + str(len(moved)) + ' files:\n' + '\n'.join(moved)

def check_website(cfg, url):
    ok, info = check_url(url, cfg)
    if not ok:
        audit('blocked_url', url)
        return 'BLOCKED: ' + info
    resp = requests.get(info, timeout=15)
    audit('check_website', info)

def output_view(text=''):
    scroll = ScrollView()
    lbl = Label(text=text, color=(0.78, 0.78, 0.78, 1),
                size_hint=(1, None), font_size='13sp', valign='top')
    lbl.text_size = (Window.width - 60, None)
    lbl.bind(texture_size=lbl.setter('size'))
    scroll.add_widget(lbl)
    return scroll, lbl
